model = ExtAPI.DataModel.Project.Model
geo = model.Geometry
connections = ExtAPI.DataModel.Project.Model.Connections

bodies = list(geo.GetChildren(DataModelObjectCategory.Body, True))

def get_body_by_name(name):
    for body in bodies:
        short_name = body.Name.split('\\')[-1]
        if short_name == name:
            return body
    return None

def make_selection(geo_body):
    sel = ExtAPI.SelectionManager.CreateSelectionInfo(SelectionTypeEnum.GeometryEntities)
    sel.Entities = [geo_body]
    return sel

body_couche1B = get_body_by_name("couche1B_S001")
body_plateau = get_body_by_name("Solid")

print("Setup OK :", len(bodies), "corps trouvés")
print(body_couche1B, body_plateau)
