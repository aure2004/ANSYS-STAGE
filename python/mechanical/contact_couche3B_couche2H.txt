from Ansys.Core.Units import Quantity

group_c3b_c2h = connections.AddConnectionGroup()
group_c3b_c2h.Name = "Couche3B-Couche2H_Contacts"
created = 0
failed = []

for j in range(1, 101):
    seg = "S{0:03d}".format(j)
    name_c3b = "couche3B_{0}".format(seg)
    name_c2h = "couche2H_{0}".format(seg)
    body_c3b = get_body_by_name(name_c3b)
    body_c2h = get_body_by_name(name_c2h)
    if body_c3b is None or body_c2h is None:
        failed.append((name_c3b, name_c2h))
        continue
    contact = group_c3b_c2h.AddContactRegion()
    contact.SourceLocation = make_selection(body_c3b.GetGeoBody())
    contact.TargetLocation = make_selection(body_c2h.GetGeoBody())
    contact.ContactType = ContactType.Bonded
    contact.AutomaticThermalConductance = False
    contact.ThermalConductanceValue = Quantity(800, "W m^-2 C^-1")
    created += 1

print("Créés :", created)
print("Échecs :", len(failed), failed)
