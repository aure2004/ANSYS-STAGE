from Ansys.Core.Units import Quantity

group_c2b_c1h = connections.AddConnectionGroup()
group_c2b_c1h.Name = "Couche2B-Couche1H_Contacts"

created = 0
failed = []

for j in range(1, 101):
    seg = "S{0:03d}".format(j)
    name_c2b = "couche2B_{0}".format(seg)
    name_c1h = "couche1H_{0}".format(seg)
    body_c2b = get_body_by_name(name_c2b)
    body_c1h = get_body_by_name(name_c1h)

    if body_c2b is None or body_c1h is None:
        failed.append((name_c2b, name_c1h))
        continue

    contact = group_c2b_c1h.AddContactRegion()
    contact.SourceLocation = make_selection(body_c2b.GetGeoBody())
    contact.TargetLocation = make_selection(body_c1h.GetGeoBody())
    contact.ContactType = ContactType.Bonded
    contact.AutomaticThermalConductance = False
    contact.ThermalConductanceValue = Quantity(800, "W m^-2 C^-1")
    created += 1

print("Créés :", created)
print("Échecs :", len(failed), failed)
